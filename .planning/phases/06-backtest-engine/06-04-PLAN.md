    ```python
    from itertools import product as itertools_product
    import optuna
    from typing import Any, Callable

    @dataclass
    class OptimizationTrial:
        params: dict
        metrics: dict
        metric_value: float  # The target metric value used for ranking

    class GridSearchOptimizer:
        """Exhaustive parameter search for small parameter spaces.

        Uses itertools.product for all combinations.
        """

        def __init__(
            self,
            feature_engine: FeatureEngine,
            risk_engine: RiskEngine,
            cost_model: CostModel,
            initial_capital: float = 1_000_000.0,
        ):
            self.feature_engine = feature_engine
            self.risk_engine = risk_engine
            self.cost_model = cost_model
            self.initial_capital = initial_capital

        def optimize(
            self,
            strategy_factory: Callable[[dict], BaseStrategy],
            ohlcv: pd.DataFrame,
            param_grid: dict[str, list],
            metric: str = "sharpe_ratio",
        ) -> list[OptimizationTrial]:
            """Run exhaustive grid search.

            Args:
                strategy_factory: Callable that takes a params dict and returns a configured BaseStrategy.
                ohlcv: Historical OHLCV data.
                param_grid: Dict mapping param names to lists of values, e.g. {"period": [10, 20, 30]}.
                metric: Metric name from BacktestResult.metrics to optimize (default: sharpe_ratio).

            Returns:
                List of OptimizationTrial sorted by metric_value descending.
            """
    ```

    Implementation:
    - Use `itertools.product(*param_grid.values())` to generate all combinations
    - Map each combination back to param names using `dict(zip(param_grid.keys(), combo))`
    - For each combo: create strategy via factory, create BacktestRunner, run backtest, extract metric
    - Sort results by metric_value descending
    - Return list of OptimizationTrial

    ```python
    class BayesianOptimizer:
        """Bayesian optimization using Optuna TPE sampler for larger parameter spaces."""

        def __init__(
            self,
            feature_engine: FeatureEngine,
            risk_engine: RiskEngine,
            cost_model: CostModel,
            initial_capital: float = 1_000_000.0,
        ):
            self.feature_engine = feature_engine
            self.risk_engine = risk_engine
            self.cost_model = cost_model
            self.initial_capital = initial_capital

        def optimize(
            self,
            strategy_factory: Callable[[dict], BaseStrategy],
            ohlcv: pd.DataFrame,
            param_space: dict[str, tuple[Any, Any, str]],
            n_trials: int = 100,
            metric: str = "sharpe_ratio",
            seed: int = 42,
        ) -> list[OptimizationTrial]:
            """Run Bayesian optimization with Optuna TPE sampler.

            Args:
                strategy_factory: Callable that takes a params dict and returns a BaseStrategy.
                ohlcv: Historical OHLCV data.
                param_space: Dict mapping param names to (low, high, type) tuples.
                    type is "int" or "float". E.g. {"period": (5, 60, "int"), "threshold": (0.5, 0.9, "float")}
                n_trials: Number of optimization trials.
                metric: Metric to maximize.
                seed: Random seed for reproducibility.

            Returns:
                List of OptimizationTrial sorted by metric_value descending.
            """
    ```

    Implementation:
    - Create `optuna.create_study(direction="maximize", sampler=optuna.samplers.TPESampler(seed=seed))`
    - Suppress Optuna's default logging: `optuna.logging.set_verbosity(optuna.logging.WARNING)`
    - Define objective function:
      - For each param in param_space: use `trial.suggest_int(name, low, high)` or `trial.suggest_float(name, low, high)` based on type
      - Create strategy, create runner, run backtest
      - Store params and full metrics as trial user_attrs
      - Return metric value
    - Call `study.optimize(objective, n_trials=n_trials)`
    - Extract results from study.trials, build OptimizationTrial list
    - Sort by metric_value descending
